### Pseudocode

START

DECLARE name as string
DECLARE rollNo as integer
DECLARE age as integer
DECLARE height as float
DECLARE GPA as float
DECLARE section as character

INPUT name

INPUT age
INPUT height
INPUT GPA
INPUT section

DISPLAY name

DISPLAY age
DISPLAY height
DISPLAY GPA
DISPLAY section

END
